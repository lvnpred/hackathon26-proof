```js
"use strict";
// Pure row-state logic for the medication rows (no DOM, no network): the page, the tests (node) and the
// summary bar all read the same rules. One predicate matters most: a row "needs you" exactly when the
// server's save gate (core/safety.py screen_unverified) would block it, i.e. status is not "verified"
// and the doctor has not confirmed it. Everything else here is wording and icon choice.
(function (root) {
  const str = (v) => (typeof v === "string" ? v : v == null ? "" : String(v));
  const same = (a, b) => str(a).trim().toLowerCase() === str(b).trim().toLowerCase();
  const DOSE_STATUS = ["unverified-strength", "dose-check"];
  const NEEDS = ["dose", "confirm"];
  const ICON = { ok: "check-circle", dose: "capsule", confirm: "triangle", you: "shield-check", known: "book" };
  const LABEL = { ok: "Matched", dose: "Check dose", confirm: "Confirm medicine", you: "Confirmed by you", known: "Known to your clinic" };
  const META_KEYS = ["heard", "lookalike", "generic", "usual_strength", "dose_note", "source"];
  const MAX_CHIPS = 8;
  // The extra row fields the server sends (contract 4.1), with safe defaults.
  function metaOf(r) {
    r = r || {};
    const m = {};
    META_KEYS.forEach((k) => { m[k] = str(r[k]); });
    m.strength_options = Array.isArray(r.strength_options) ? r.strength_options.map(str).filter(Boolean) : [];
    return m;
  }
  function chipsFor(row) {
    const usual = str(row.usual_strength);
    const opts = Array.isArray(row.strength_options) ? row.strength_options.map(str).filter(Boolean) : [];
    const seen = new Set();
    const list = [];
    opts.forEach((o) => { const k = o.toLowerCase(); if (!seen.has(k)) { seen.add(k); list.push(o); } });
    list.sort((a, b) => (same(b, usual) ? 1 : 0) - (same(a, usual) ? 1 : 0));   // stable: usual first, rest keep order
    return list.slice(0, MAX_CHIPS).map((label) => ({ label, usual: usual !== "" && same(label, usual) }));
  }
  function reasonFor(row, state, ctx) {
    const status = str(row.status);
    const name = str(row.name).trim();
    const heard = str(row.heard).trim();
    const strength = str(ctx.strength != null ? ctx.strength : row.strength).trim();
    const note = str(row.dose_note).trim();
    if (state === "ok") return "Name and strength match the medicine list.";
    if (state === "known") return "Known to your clinic from earlier prescriptions. There is no dose reference for it yet.";
    if (state === "you") {
      const how = str(ctx.how);
      const base = how === "spell" ? "You chose this name." : how === "chip" ? "You chose the strength " + strength + "." : "You accepted this row as written.";
      return note ? base + " Note: " + note : base;
    }
    if (state === "dose") {
      if (status === "dose-check") return note || "The daily total looks higher than usual. Check the dose.";
      const tap = Array.isArray(row.strength_options) && row.strength_options.length ? " Tap one below." : "";
      return (strength ? "Strength " + strength + " is not one this medicine usually comes in." : "No strength was dictated.") + tap;
    }
    if (status === "lookalike-check") {
      const twin = str(row.lookalike).trim();
      return (heard ? 'Heard "' + heard + '". ' : "") + (twin ? "It looks like " + twin + ". They are different medicines." : "Another medicine has a very similar name.");
    }
    if (status === "unknown-medicine") return '"' + (heard || name) + '" is not in the medicine list.';
    if (status === "unverified-name") return (heard ? 'Heard "' + heard + '". ' : "") + "The name was not clearly dictated.";
    if (status === "") return "You changed this row. Confirm it before saving.";
    return "This row needs a check (status: " + status + ").";
  }
  // row: the Rx row fields (status, source, heard, lookalike, dose_note, strength_options, usual_strength, ...).
  // ctx: { confirmed: bool, strength: current strength text, how: "confirm" | "spell" | "chip" }.
  function rowState(row, ctx) {
    row = row || {};
    ctx = ctx || {};
    const status = row.status == null ? "" : str(row.status);
    let state;
    if (ctx.confirmed) state = "you";
    else if (status === "verified") state = str(row.source) === "clinic" ? "known" : "ok";
    else if (DOSE_STATUS.indexOf(status) >= 0) state = "dose";
    else state = "confirm";                       // name statuses, "" (edited) and any future status: the safe side
    const edited = state === "confirm" && status === "";
    return {
      state,
      icon: ICON[state],
      label: edited ? "Confirm row" : LABEL[state],
      reason: reasonFor(Object.assign({}, row, { status }), state, ctx),
      needsYou: NEEDS.indexOf(state) >= 0,
      canConfirm: NEEDS.indexOf(state) >= 0,
      canSpell: state === "confirm",
      canReopen: state === "you",
      chips: state === "dose" ? chipsFor(row) : [],
    };
  }
  // After an edit by hand the old reasons are stale: drop them (a name edit also drops what belonged to the old name).
  function afterEdit(meta, col) {
    const m = Object.assign({}, meta);
    m.heard = ""; m.lookalike = ""; m.dose_note = "";
    if (col === "name") { m.generic = ""; m.source = ""; m.usual_strength = ""; m.strength_options = []; }
    return m;
  }
  // counts: {total, needs, checking} -> the summary bar words and the Save button.
  function summary(counts) {
    const total = counts.total | 0, needs = counts.needs | 0, checking = counts.checking | 0;
    return {
      title: total + (total === 1 ? " medicine" : " medicines"),
      ready: (total - needs) + " ready",
      needs: needs + (needs === 1 ? " needs you" : " need you"),
      saveLabel: checking ? "Checking..." : needs ? "Resolve " + needs + " to save" : "Save prescription",
      saveDisabled: checking > 0 || needs > 0,
    };
  }
  root.RxState = { rowState, metaOf, chipsFor, afterEdit, summary, NEEDS };
})(typeof window !== "undefined" ? window : globalThis);
```

// The medication rows on screen: state cell, strength chips, generic line, summary bar. Rules live in rxstate.js;
// app.js hands over what it owns once, through RxView.init(hooks). DOM is built with createElement and textContent only.
window.RxView = (function () {
  const NS = "http://www.w3.org/2000/svg", COLS = ["form", "name", "strength", "dose", "timing", "duration", "instructions"];
  const ICONS = {
    "check-circle": [["circle", { cx: 12, cy: 12, r: 9 }], ["path", { d: "M8 12.5l2.7 2.7L16 9.5" }]],
    capsule: [["rect", { x: 3, y: 9, width: 18, height: 6, rx: 3, transform: "rotate(-35 12 12)" }], ["path", { d: "M9.2 7.2l5.6 9.6" }]],
    triangle: [["path", { d: "M12 4l9 16H3z" }], ["path", { d: "M12 10v4M12 17.2v.1" }]],
    "shield-check": [["path", { d: "M12 3l7 3v5c0 5-3 8-7 10-4-2-7-5-7-10V6z" }], ["path", { d: "M9 12l2.2 2.2L15.5 10" }]],
    book: [["path", { d: "M5 4h10a3 3 0 013 3v13H8a3 3 0 01-3-3z" }], ["path", { d: "M5 17a3 3 0 013-3h10" }]],
  };
  let H = null, seq = 0;
  const $ = (id) => document.getElementById(id);
  const val = (tr, c) => { const n = tr.querySelector('[data-col="' + c + '"]'); return (n ? n.textContent : "").replace(/\s+/g, " ").trim(); };
  const rows = () => [...H.body.querySelectorAll("tr[data-index]")];
  const isBlank = (tr) => COLS.every((c) => !val(tr, c));
  function el(tag, cls, t) { const n = document.createElement(tag); if (cls) n.className = cls; if (t != null) n.textContent = t; return n; }
  function btn(cls, action, t) { const b = el("button", "rx-act " + cls, t); b.type = "button"; b.dataset.action = action; return b; }
  function icon(name, size) {
    const svg = document.createElementNS(NS, "svg");
    const put = (n, a) => Object.keys(a).forEach((k) => n.setAttribute(k, a[k]));
    put(svg, { width: size || 16, height: size || 16, viewBox: "0 0 24 24", fill: "none", stroke: "currentColor", "stroke-width": 2.4, "stroke-linecap": "round", "stroke-linejoin": "round", "aria-hidden": "true" });
    (ICONS[name] || []).forEach((p) => { const n = document.createElementNS(NS, p[0]); put(n, p[1]); svg.appendChild(n); });
    return svg;
  }
  const fresh = (src) => ({ meta: RxState.metaOf(src), how: (src && src.how) || "", note: "", req: ++seq });
  const rx = (tr) => tr._rx || (tr._rx = fresh(null));
  const rowOf = (tr) => Object.assign({ status: tr.dataset.status || "", name: val(tr, "name"), strength: val(tr, "strength") }, rx(tr).meta);
  const ctxOf = (tr) => ({ confirmed: H.confirmed().has(tr.dataset.rowid), strength: val(tr, "strength"), how: rx(tr).how });
  function renderRow(tr) {
    if (!H) return;
    const r = rx(tr), st = RxState.rowState(rowOf(tr), ctxOf(tr)), name = val(tr, "name") || "this row";
    tr.dataset.rxstate = st.state;
    const med = tr.querySelector("td.med");
    med.querySelectorAll(".rx-extra").forEach((n) => n.remove());                       // generic line and chips: screen only
    if (r.meta.generic) med.appendChild(el("div", "rx-extra rx-sub no-print", "generic: " + r.meta.generic));
    if (st.chips.length) {
      const box = el("div", "rx-extra rx-chips no-print");
      box.appendChild(el("span", "rx-hint", "Strengths:"));
      st.chips.forEach((c) => {
        const b = btn("rx-chip" + (c.usual ? " usual" : ""), "chip", c.label);

        if (c.usual) b.appendChild(el("span", "rx-usual", "usual"));
        box.appendChild(b);
      });
      med.appendChild(box);
    }
    const cell = tr.querySelector("td.check");
    let box = cell.querySelector(".rx-state");
    if (!box) {                                   // built once, so the role=status line stays one live region
      cell.textContent = "";
      box = el("div", "rx-state no-print"); box.tabIndex = -1;
      box.append(el("span", "rx-pill"), el("div", "rx-reason"), el("div", "rx-note"), el("div", "rx-acts"));
      box.children[1].setAttribute("role", "status"); box.children[2].setAttribute("role", "alert");
      cell.append(box);
    }
    box.children[0].dataset.state = st.state; box.children[0].title = st.reason; box.children[0].textContent = "";
    box.children[0].append(icon(st.icon), el("span", null, st.label));
    box.children[1].textContent = st.reason; box.children[2].textContent = r.note;
    const acts = box.children[3]; acts.textContent = "";
    if (st.canConfirm) { const b = btn("mini solid confirm-btn", "confirm", "Confirm as is"); b.title = "Accept this medicine exactly as written"; b.setAttribute("aria-label", "Confirm " + name + " as is"); acts.appendChild(b); }
    if (st.canReopen) { const b = btn("mini confirm-btn", "confirm", "Reopen"); b.title = "Reopen this row for checking"; acts.appendChild(b); }
  }
  function sync() {                                // the summary bar and the Save button, from the same rules as the rows
    if (!H) return;
    let needs = 0;
    const live = rows().filter((tr) => !isBlank(tr));
    live.forEach((tr) => { if (RxState.rowState(rowOf(tr), ctxOf(tr)).needsYou) needs++; });
    const s = RxState.summary({ total: live.length, needs, checking: 0 });
    $("rxBar").hidden = !H.hasSheet();
    $("rxTitle").textContent = s.title;
    $("rxReady").lastChild.textContent = s.ready; $("rxNeeds").lastChild.textContent = s.needs;
    $("rxNeeds").dataset.state = needs ? "dose" : "ok";
    H.saveBtn.textContent = s.saveLabel;
    H.saveBtn.disabled = H.working() || s.saveDisabled || !!(window.Authorship && Authorship.isViewOnly());
  }
  function pickChip(tr, value) {                   // a tap is the doctor's choice: set the cell, mark the row confirmed
    if (H.working()) return;
    const n = tr.querySelector('[data-col="strength"]'); if (n) n.textContent = value;
    rx(tr).how = "chip"; H.confirmed().add(tr.dataset.rowid);
    renderRow(tr); H.changed();
    const s = tr.querySelector(".rx-state"); if (s) s.focus();
  }
  function onClick(e) {
    const b = e.target && e.target.closest ? e.target.closest('button[data-action="chip"]') : null;
    if (b && !(window.Authorship && Authorship.isViewOnly())) pickChip(b.closest("tr[data-index]"), b.dataset.value);
  }
  function init(hooks) {
    H = hooks;
    H.body.addEventListener("click", onClick);
    [["rxReady", "check-circle"], ["rxNeeds", "triangle"]].forEach((p) => $(p[0]).append(icon(p[1], 14), el("span")));
  }
  // app.js calls these: attach() when a row is built, onEdit() after a hand edit, meta() for the undo snapshot.
  function attach(tr, src) { tr._rx = fresh(src); }
  function onEdit(tr, target) {
    const c = target && target.closest ? target.closest("[data-col]") : null, r = rx(tr);
    r.meta = RxState.afterEdit(r.meta, c ? c.dataset.col : ""); r.how = ""; r.note = "";
  }
  const meta = (tr) => Object.assign({}, rx(tr).meta, { how: rx(tr).how });
  return { init, attach, renderRow, sync, onEdit, meta, isBlank };
})();
